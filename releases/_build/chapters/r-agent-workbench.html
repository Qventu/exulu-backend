const res = await fetch("https://exulu.your-company.com/agents/litellm/run/agt_8fk2", {
  method: "POST",
  headers: {
    "content-type": "application/json",
    authorization: `Bearer ${process.env.EXULU_API_TOKEN}`,
    session: "YOUR_SESSION_ID",
    stream: "true",
  },
  body: JSON.stringify({
    message: {
      parts: [{ type: "text", text: "Hello" }],
      role: "user",
    },
  }),
});

// The response is a streamed UI-message stream.
for await (const chunk of res.body!.pipeThrough(new TextDecoderStream())) {
  console.log(chunk);
}
